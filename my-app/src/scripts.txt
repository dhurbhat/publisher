/* src/client.js */
let activeChapterSlug = null;
let selectedSentenceId = null;

function toggleSidebar() {
  const sidebar = document.getElementById('left-sidebar');
  sidebar.classList.toggle('collapsed');
}

function initializeAuth() {
  const rawLocationSearch = window.location.search;
  const tokenMatch = rawLocationSearch.match(/token=([^&]+)/);
  const authStatus = document.getElementById('auth-status');
  
  if (tokenMatch && tokenMatch[1]) {
    localStorage.setItem('ucan_token', decodeURIComponent(tokenMatch[1]));
    window.history.replaceState({}, document.title, window.location.pathname);
  }
  
  const storedToken = localStorage.getItem('ucan_token');
  if (storedToken) {
    authStatus.innerText = "✓ Token Active";
    authStatus.style.borderColor = "#10b981";
    authStatus.style.color = "#10b981";
    loadChapterCatalog();
  } else {
    authStatus.innerText = "No Access Token";
    authStatus.style.borderColor = "#ef4444";
    authStatus.style.color = "#ef4444";
  }
}
// NEW: Dynamic sidebar listing logic engine
async function loadChapterCatalog() {
  const token = localStorage.getItem('ucan_token');
  const listElement = document.getElementById('dynamic-chapter-list');
  if (!token || !listElement) return;

  try {
    const response = await fetch('/api/chapters', {
      method: 'GET',
      headers: { 'Authorization': 'Bearer ' + token }
    });

    if (!response.ok) throw new Error();
    const data = await response.json();
    
    if (data.chapters.length === 0) {
      listElement.innerHTML = '<li style="color: #6b7280; font-size: 0.9rem;">No manuscript chapters uploaded yet.</li>';
      return;
    }

    // Sort chapters naturally by name string patterns (handles numeric prefixes correctly)
    const sortedChapters = data.chapters.sort((a, b) => a.localeCompare(b, undefined, { numeric: true, sensitivity: 'base' }));

    // Generate dynamic links
    listElement.innerHTML = sortedChapters.map(slug => {
      // Turn slug patterns like "chapter-1-the-sos-event" into clean display names like "Chapter 1 The Sos Event"
      let displayName = slug
        .replace('chapter-', '')
        .replace(/-/g, ' ');
      displayName = displayName.replace(/^\d+[\s_.-]*/,'');
      displayName = displayName.replace(/\b\w/g, c => c.toUpperCase()); // Capitalize words cleanly
        
      return `
        <li style="margin-bottom: 8px;">
          <a href="#" onclick="event.preventDefault(); loadChapter('${slug}')" style="text-decoration: none; color: #4b5563; font-size: 0.95rem; display: block; padding: 4px 0;" onmouseover="this.style.color='#111827'" onmouseout="this.style.color='#4b5563'">
            ${displayName}
          </a>
        </li>
      `;
    }).join('');

  } catch (error) {
    listElement.innerHTML = '<li style="color: #ef4444; font-size: 0.9rem;">Failed to load index catalog.</li>';
  }
}

function parseSentencesIntoSpans(text) {
  // 1. Separate block text into structural paragraphs by splitting line-breaks
  const paragraphs = text.split(/\n+/);
  let globalIndexCounter = 0;

  // Initialize the native browser sentence-splitting engine
  const segmenter = new Intl.Segmenter(undefined, { granularity: 'sentence' });

  return paragraphs.map(paragraphContent => {
    const trimmed = paragraphContent.trim();
    if (!trimmed) return '';

    // Handle your Markdown headers cleanly
    if (trimmed.startsWith('# ')) {
      return `<h1 style="margin: 2rem 0 1rem 0; font-size: 1.8rem; font-weight: 700; color: #111827;">${trimmed.substring(2)}</h1>`;
    }
    if (trimmed.startsWith('## ')) {
      return `<h2 class="manuscript-section" style="margin: 1.5rem 0 0.75rem 0; font-size: 1.4rem; font-weight: 600; color: #374151;">${trimmed.substring(3)}</h2>`;
    }

    // 2. NATIVE SAFARI SPLITTING: Let the browser natively slice the text safely!
    const segments = segmenter.segment(trimmed);
    const wrappedSpans = Array.from(segments).map(segmentObj => {
      const sentenceText = segmentObj.segment;
      if (!sentenceText.trim()) return '';

      const currentId = globalIndexCounter++;
      return `<span class="novel-sentence" data-id="${currentId}" onclick="event.stopPropagation(); window.selectSentence(this, ${currentId})">${sentenceText}</span>`;
    }).join('');

    return `<p style="margin-bottom: 1.5rem; text-indent: 1.5rem; text-align: justify; font-size: 1.15rem; line-height: 1.8;">${wrappedSpans}</p>`;
  }).join('');
}

async function loadChapter(slug) {
  const canvas = document.getElementById('text-canvas');
  const token = localStorage.getItem('ucan_token');
  if (!token) {
    alert('Access Denied: No valid token found.');
    return;
  }
  
  activeChapterSlug = slug;
  canvas.innerHTML = '<p style="color: #6b7280; text-align: center;">Securely loading chapter...</p>';

  try {
    const response = await fetch('/api/chapter/' + slug, {
      method: 'GET',
      headers: { 'Authorization': 'Bearer ' + token }
    });
    if (!response.ok) throw new Error();
    
    const chapterText = await response.text();

    canvas.innerHTML = `
      <h1 style="margin-bottom: 28px; font-size: 1.8rem; font-weight: 700;">${slug.toUpperCase().replace('-', ' ')}</h1>
      <div id="manuscript-viewport" style="line-height: 1.85; font-size: 1.15rem;">${parseSentencesIntoSpans(chapterText)}</div>
    `;
    buildSubsectionNavigation();
    resetFeedbackUI();
  } catch (error) {
    canvas.innerHTML = '<h1 style="color: #ef4444; text-align: center;">Error Loading Content</h1>';
  }
}
function buildSubsectionNavigation() {
  const sections = document.querySelectorAll('.manuscript-section');
  const navContainer = document.getElementById('subsection-nav-container');
  const linksList = document.getElementById('subsection-links');
  linksList.innerHTML = '';

  if (sections.length === 0) {
    navContainer.style.display = 'none';
    return;
  }
  
  sections.forEach((section, index) => {
    const sectionId = `section-anchor-${index}`;
    section.id = sectionId;
    const li = document.createElement('li');
    li.innerHTML = `<a href="#${sectionId}" onclick="event.preventDefault(); document.getElementById('${sectionId}').scrollIntoView({ behavior: 'smooth' });" style="text-decoration: none; color: #4b5563; transition: color 0.15s;" onmouseover="this.style.color='#111827'" onmouseout="this.style.color='#4b5563'">${section.innerText}</a>`;
    linksList.appendChild(li);
  });
  navContainer.style.display = 'block';
}

window.selectSentence = function(element, id) {
  document.querySelectorAll('.novel-sentence').forEach(el => el.classList.remove('active-highlight'));
  element.classList.add('active-highlight');
  selectedSentenceId = id;

  const panel = document.getElementById('comment-stream');
  panel.innerHTML = `
    <h4 style="margin-bottom: 12px; font-size: 0.95rem; color: #111827;">Leave Note for Sentence #${id}</h4>
    <textarea id="feedback-note" style="width: 100%; height: 120px; padding: 10px; border: 1px solid #d1d5db; border-radius: 6px; font-family: inherit; margin-bottom: 12px; resize: none;" placeholder="Type your edits or critiques here..."></textarea>
    <button onclick="submitLineNote()" style="width: 100%; background: #111827; color: white; padding: 10px; border: none; border-radius: 6px; font-weight: 500; cursor: pointer;">Save Review Note</button>
  `;
}

window.resetFeedbackUI = function() {
  selectedSentenceId = null;
  document.getElementById('comment-stream').innerHTML = '<div style="color: #9ca3af; font-size: 0.9rem;">Click on any line inside the text canvas to view or drop inline notes</div>';
}

window.submitLineNote = async function() {
  const commentText = document.getElementById('feedback-note').value;
  const token = localStorage.getItem('ucan_token');
  const panel = document.getElementById('comment-stream');

  if (!commentText.trim()) {
    alert('Please type a feedback comment before saving.');
    return;
  }

  try {
    const response = await fetch('/api/feedback', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer ' + token
      },
      body: JSON.stringify({ 
        chapterSlug: activeChapterSlug, 
        sentenceId: selectedSentenceId,
        feedbackText: commentText 
      })
    });

    if (!response.ok) throw new Error();
    alert('Note pinned successfully!');
    resetFeedbackUI(); 
  } catch (error) {
    alert('Network error while saving feedback.');
  }
}

window.onload = initializeAuth;
